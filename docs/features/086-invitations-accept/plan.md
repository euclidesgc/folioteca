      ```sql
      ALTER TABLE "Invitation" ADD COLUMN "acceptedAt" TIMESTAMP(3);

      -- with the 0009 index, an address that already accepted an invitation could
      -- never be invited again. The index becomes partial over the pending rows.

      ```
